# 安裝PyTorch（需要先安裝，因為vllm依賴它）
pip install torch==${TORCH}+${CUDA} \
            torchvision \
            torchaudio \
            --index-url https://download.pytorch.org/whl/${CUDA}

# 先確認實際 ABI，再挑對應的 flash_attn wheel（避免 undefined symbol 錯誤）
ABI=$(python -c "import torch; print('TRUE' if torch._C._GLIBCXX_USE_CXX11_ABI else 'FALSE')")
echo "偵測到目前 torch 的 CXX11 ABI = ${ABI}"

FLASH_ATTN_WHL="flash_attn-2.8.2+cu12torch${TORCH_MINOR}cxx11abi${ABI}-cp310-cp310-linux_x86_64.whl"
wget https://github.com/Dao-AILab/flash-attention/releases/download/v2.8.2/${FLASH_ATTN_WHL}
pip install ./${FLASH_ATTN_WHL}
